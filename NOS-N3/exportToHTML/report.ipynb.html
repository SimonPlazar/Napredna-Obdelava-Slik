import torch


def iou_score(pred, target, eps=1e-7):
    """ 
    pred, target: (B, 1, H, W), binary {0,1} 
    """
    tp = (pred & target).sum(dim=(1, 2, 3))
    fp = (pred & (~target)).sum(dim=(1, 2, 3))
    fn = ((~pred) & target).sum(dim=(1, 2, 3))

    iou = tp / (tp + fp + fn + eps)
    return iou


def dice_score(pred, target, eps=1e-7):
    """ 
    pred, target: (B, 1, H, W), binary {0,1} 
    """
    tp = (pred & target).sum(dim=(1, 2, 3))
    fp = (pred & (~target)).sum(dim=(1, 2, 3))
    fn = ((~pred) & target).sum(dim=(1, 2, 3))

    dice = (2 * tp) / (2 * tp + fp + fn + eps)
    return dice

from torch.utils.data import DataLoader

NUM_TEST_IMAGES = 1000
BATCH_SIZE = 16

dataset_random = TextSegmentationDataset(
    image_dir=DATASET_PATH_TEST,
    checkpoint_dir='checkpoints/dataset_random',
    length=NUM_TEST_IMAGES,
    mode='random'
)

test_loader = DataLoader(
    dataset_random,
    batch_size=BATCH_SIZE,
    shuffle=False,
    num_workers=0
)
#%% 
from tqdm import tqdm

model.eval()

all_ious = []
all_dices = []

with torch.no_grad():
    for images, masks in tqdm(test_loader, desc="Testing", unit="batch"):
        images = images.to(DEVICE)
        masks = masks.to(DEVICE).int()

        logits = model(images)
        probs = torch.sigmoid(logits)
        preds = (probs > 0.5).int()

        iou = iou_score(preds, masks)
        dice = dice_score(preds, masks)

        all_ious.extend(iou.cpu().numpy())
        all_dices.extend(dice.cpu().numpy())

iou_mean = np.mean(all_ious)
iou_std = np.std(all_ious)

dice_mean = np.mean(all_dices)
dice_std = np.std(all_dices)

print("\n=== TEST RESULTS (1000 RANDOM IMAGES) ===")
print(f"IoU  : mean = {iou_mean:.4f}, std = {iou_std:.4f}")
print(f"Dice : mean = {dice_mean:.4f}, std = {dice_std:.4f}")

import random


def visualize_random_examples(model, dataset, num_samples=3):
    model.eval()
    indices = random.sample(range(len(dataset)), num_samples)

    plt.figure(figsize=(16, 4 * num_samples))

    with torch.no_grad():
        for i, idx in enumerate(indices):
            image, mask = dataset[idx]

            image_t = image.unsqueeze(0).to(DEVICE)
            logits = model(image_t)
            pred = (torch.sigmoid(logits)[0, 0] > 0.5).cpu()

            image_np = image.permute(1, 2, 0).cpu().numpy()
            mask_np = mask[0].cpu().numpy()
            pred_np = pred.numpy()

            # Calculate difference (errors)
            diff = np.abs(mask_np.astype(float) - pred_np.astype(float))

            # Input image
            plt.subplot(num_samples, 4, i * 4 + 1)
            plt.imshow(image_np)
            plt.title("Input image")
            plt.axis("off")

            # Ground truth
            plt.subplot(num_samples, 4, i * 4 + 2)
            plt.imshow(mask_np, cmap="gray")
            plt.title("Ground truth mask")
            plt.axis("off")

            # Prediction
            plt.subplot(num_samples, 4, i * 4 + 3)
            plt.imshow(pred_np, cmap="gray")
            plt.title("Predicted mask")
            plt.axis("off")

            # Difference
            plt.subplot(num_samples, 4, i * 4 + 4)
            plt.imshow(diff, cmap="Reds")
            plt.title("Difference (errors)")
            plt.axis("off")

    plt.tight_layout()
    plt.show()


visualize_random_examples(model, dataset_random, num_samples=3)
