def apply_predicted_region(masked_img, predicted_img, mask):
    result = masked_img.clone()
    # Expand mask to match channel dimension [1, 1, H, W] -> [1, 3, H, W]
    mask_expanded = mask.expand_as(result)
    # Where mask == 0 (masked area), use predicted; where mask == 1, keep original
    result = torch.where(mask_expanded == 0, predicted_img, result)
    return result

# Statistično testiranje na 1000 slikah
print("\n=== Statistical Testing on 1000 Images ===\n")

# Metrics storage
psnr_values = []
ssim_values = []
l1_values = []
l2_values = []

with torch.no_grad():
    for i in range(1000):
        if (i + 1) % 100 == 0:
            print(f"Processing image {i + 1}/1000...")

        masked_img, mask, original_img = dataset_test[i]

        # Move to device
        masked_input = masked_img.unsqueeze(0).to(device)
        mask_input = mask.unsqueeze(0).to(device)
        original_input = original_img.unsqueeze(0).to(device)

        # Generate inpainted image
        output = model(masked_input, mask_input)

        # Calculate metrics
        # L1 Loss
        l1_loss = F.l1_loss(output, original_input).item()
        l1_values.append(l1_loss)

        # L2 Loss (MSE)
        l2_loss = F.mse_loss(output, original_input).item()
        l2_values.append(l2_loss)

        # Convert to numpy for PSNR and SSIM
        output_np = output.squeeze(0).cpu().permute(1, 2, 0).numpy()
        original_np = original_img.permute(1, 2, 0).numpy()

        # PSNR
        psnr_val = psnr(original_np, output_np, data_range=1.0)
        psnr_values.append(psnr_val)

        # SSIM
        ssim_val = ssim(original_np, output_np, data_range=1.0, channel_axis=2)
        ssim_values.append(ssim_val)

# Calculate statistics
metrics = {
    'PSNR (dB)': psnr_values,
    'SSIM': ssim_values,
    'L1 Loss': l1_values,
    'L2 Loss (MSE)': l2_values
}

print("\n=== Evaluation Results ===\n")
print(f"{'Metric':<15} {'Mean':<12} {'Std Dev':<12}")
print("-" * 40)

for metric_name, values in metrics.items():
    mean_val = np.mean(values)
    std_val = np.std(values)
    print(f"{metric_name:<15} {mean_val:<12.4f} {std_val:<12.4f}")

# Visualize metric distributions
fig, axes = plt.subplots(2, 2, figsize=(14, 10))
axes = axes.flatten()

for idx, (metric_name, values) in enumerate(metrics.items()):
    axes[idx].hist(values, bins=50, edgecolor='black', alpha=0.7)
    axes[idx].axvline(np.mean(values), color='red', linestyle='--',
                      linewidth=2, label=f'Mean: {np.mean(values):.4f}')
    axes[idx].set_xlabel(metric_name)
    axes[idx].set_ylabel('Frequency')
    axes[idx].set_title(f'{metric_name} Distribution')
    axes[idx].legend()
    axes[idx].grid(True, alpha=0.3)

plt.tight_layout()
plt.show()
