#%% 
from pathlib import Path
import time

from torch import nn

from Model import SuperResolution, T91TrainDataset, TestDataset
from Model import calculate_psnr, calculate_ssim

from Helper import *
#%% 
# 2X SCALING MODEL

# ===== Hiperparametri učenja =====

SCALE = 3
PATCH_SIZE = 33
BATCH_SIZE = 256
NUM_EPOCHS = 5000
LEARNING_RATE = 1e-5
MIN_PATCHES = ((1000 + BATCH_SIZE - 1) // BATCH_SIZE) * BATCH_SIZE

T91_DIR = os.path.join("datasets", "T91")  # Train dataset directory
# SET_DIR = os.path.join("datasets", "Set5") # Test dataset directory (for validation)
SET_DIR = os.path.join("datasets", "Set14")  # Test dataset directory (for validation)

# Training configuration
VAL_INTERVAL = 50  # Validate every N epochs
SAVE_INTERVAL = 100  # Save checkpoint every N epochs
LOG_INTERVAL = 20  # Log training loss every N epochs

device = "cuda" if torch.cuda.is_available() else "cpu"
print("Using device:", device)

# Initialize model
model = SuperResolution().to(device)
print(f"Model parameters: {sum(p.numel() for p in model.parameters()):,}")

# Loss and optimizer
criterion = nn.MSELoss()
optimizer = torch.optim.Adam(
    model.parameters(),
    lr=LEARNING_RATE
)

# Learning rate scheduler
scheduler = torch.optim.lr_scheduler.ReduceLROnPlateau(
    optimizer,
    mode='min',
    factor=0.5,
    patience=100
)

# Training dataset
train_dataset = T91TrainDataset(
    dataset_dir=T91_DIR,
    scale_factor=SCALE,
    patch_size=PATCH_SIZE,
    min_patches=MIN_PATCHES
)
train_loader = torch.utils.data.DataLoader(
    dataset=train_dataset,
    batch_size=BATCH_SIZE,
    shuffle=True,
    num_workers=0,
    pin_memory=True,
    drop_last=True
)

# Validation dataset
val_dataset = TestDataset(
    test_dir=SET_DIR,
    scale_factor=SCALE
)

print(f"Training on {len(train_dataset.image_paths)} images ({len(train_loader)} batches per epoch)")
print(f"Validation on {len(val_dataset)} images")


#%% 
# Setup checkpoint directory

checkpoint_dir = Path(f"./checkpoints_x{SCALE}")
checkpoint_dir.mkdir(exist_ok=True)

# Load checkpoint if exists
checkpoint = load_checkpoint_generic(checkpoint_dir, device)

# Initialize tracking variables
train_losses = []
val_losses = []
val_psnrs = []
val_ssims = []
epochs_log = []
start_epoch = 1

if checkpoint:
    model.load_state_dict(checkpoint['model_state_dict'])
    optimizer.load_state_dict(checkpoint['optimizer_state_dict'])
    if 'scheduler_state_dict' in checkpoint:
        scheduler.load_state_dict(checkpoint['scheduler_state_dict'])
    start_epoch = checkpoint.get('epoch', 0) + 1
    train_losses = checkpoint.get('train_losses', [])
    val_losses = checkpoint.get('val_losses', [])
    val_psnrs = checkpoint.get('val_psnrs', [])
    val_ssims = checkpoint.get('val_ssims', [])
    epochs_log = checkpoint.get('epochs_log', [])
    print(f"Resuming training from epoch {start_epoch}")
    print(f"Loaded history: {len(val_losses)} validation points")
else:
    print("No checkpoint found, starting fresh training.")

best_val_loss = checkpoint.get('best_val_loss', float('inf')) if checkpoint else float('inf')

print(f"\nStarting training from epoch {start_epoch} to {NUM_EPOCHS}...")

#%% 
def validate_model(model, val_dataset, device):
    """Run validation on test dataset"""
    model.eval()
    total_loss = 0.0
    total_psnr = 0.0
    total_ssim = 0.0

    with torch.no_grad():
        for idx in range(len(val_dataset)):
            lr_ycbcr, hr_ycbcr = val_dataset[idx]

            # Extract Y channel
            lr_y = lr_ycbcr[:, :, 0]
            hr_y = hr_ycbcr[:, :, 0]

            # Convert to tensor [1, 1, H, W]
            lr_tensor = torch.from_numpy(lr_y).unsqueeze(0).unsqueeze(0).to(device)
            hr_tensor = torch.from_numpy(hr_y).unsqueeze(0).unsqueeze(0).to(device)

            # Forward pass
            sr_tensor = model(lr_tensor)

            # Calculate loss
            loss = criterion(sr_tensor, hr_tensor)
            total_loss += loss.item()

            # Move to CPU once for both metrics
            hr_cpu = hr_tensor.cpu()
            sr_cpu = sr_tensor.cpu()

            # Calculate PSNR and SSIM
            psnr = calculate_psnr(hr_cpu, sr_cpu)
            ssim = calculate_ssim(hr_cpu, sr_cpu)

            total_psnr += psnr
            total_ssim += ssim

    avg_loss = total_loss / len(val_dataset)
    avg_psnr = total_psnr / len(val_dataset)
    avg_ssim = total_ssim / len(val_dataset)

    model.train()
    return avg_loss, avg_psnr, avg_ssim
#%% 
# Training loop

# Time tracking
start_time = time.time()
running_loss = 0.0

model.train()

for epoch in range(start_epoch, NUM_EPOCHS + 1):
    epoch_loss = 0.0

    for y_lr, y_hr in train_loader:
        # Move to device
        y_lr = y_lr.to(device, non_blocking=True)
        y_hr = y_hr.to(device, non_blocking=True)

        # Forward pass
        optimizer.zero_grad()
        y_pred = model(y_lr)

        # Compute loss
        loss = criterion(y_pred, y_hr)

        # Backward pass
        loss.backward()
        optimizer.step()

        epoch_loss += loss.item()

    # Average loss for epoch
    avg_epoch_loss = epoch_loss / len(train_loader)
    train_losses.append(avg_epoch_loss)

    # Log progress
    if epoch % LOG_INTERVAL == 0:
        elapsed = time.time() - start_time
        print(f"Epoch {epoch:4d}/{NUM_EPOCHS} | Train Loss: {avg_epoch_loss:.6f} | Time: {elapsed:.2f}s")
        start_time = time.time()

    # Validation
    if epoch % VAL_INTERVAL == 0:
        val_loss, val_psnr, val_ssim = validate_model(model, val_dataset, device)

        val_losses.append(val_loss)
        val_psnrs.append(val_psnr)
        val_ssims.append(val_ssim)
        epochs_log.append(epoch)

        current_lr = optimizer.param_groups[0]['lr']
        print(f">>> Validation at epoch {epoch:4d} | Val Loss: {val_loss:.6f} | "
              f"PSNR: {val_psnr:.4f} dB | SSIM: {val_ssim:.4f} | LR: {current_lr:.6f}")

        scheduler.step(val_loss)

        # Save best model
        if val_loss < best_val_loss:
            best_val_loss = val_loss
            torch.save({
                'epoch': epoch,
                'model_state_dict': model.state_dict(),
                'optimizer_state_dict': optimizer.state_dict(),
                'scheduler_state_dict': scheduler.state_dict(),
                'scale': SCALE,
                'val_loss': val_loss,
                'val_psnr': val_psnr,
                'val_ssim': val_ssim,
                'lr': current_lr
            }, checkpoint_dir / 'best_model.pth')
            print(f"✓ Saved best model (Loss: {val_loss:.6f}, PSNR: {val_psnr:.4f} dB, SSIM: {val_ssim:.4f})")

    # Save checkpoint
    if epoch % SAVE_INTERVAL == 0:
        save_checkpoint_generic(
            checkpoint_dir,
            epoch,
            {
                'epoch': epoch,
                'model_state_dict': model.state_dict(),
                'optimizer_state_dict': optimizer.state_dict(),
                'scheduler_state_dict': scheduler.state_dict(),
                'scale': SCALE,
                'lr': optimizer.param_groups[0]['lr'],
                'best_val_loss': best_val_loss,
                'train_losses': train_losses,
                'val_losses': val_losses,
                'val_psnrs': val_psnrs,
                'val_ssims': val_ssims,
                'epochs_log': epochs_log
            },
            max_checkpoints=5
        )

print("\n🎉 Training completed!")
print(f"📊 Final statistics:")
print(f"  - Best validation loss: {best_val_loss:.6f}")
print(f"  - Total validation points: {len(val_losses)}")
print(f"  - Total training epochs: {len(train_losses)}")
