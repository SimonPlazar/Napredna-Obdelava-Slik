def plot_training_metrics(checkpoint_dir, save_fig=False):
    """ 
    Plot comprehensive training metrics from checkpoint data. 
 
    Args: 
        checkpoint_dir: Directory containing checkpoints 
        save_fig: If True, saves plots to file 
    """
    from pathlib import Path

    VALIDATION_INTERVAL = 100
    checkpoint_dir = Path(checkpoint_dir)
    checkpoint_dir.mkdir(parents=True, exist_ok=True)

    checkpoint = load_checkpoint_generic(checkpoint_dir, device='cpu')

    if checkpoint is None:
        print("❌ No checkpoint found!")
        return

    # Extract data
    train_losses = checkpoint.get('train_losses', [])
    val_losses = checkpoint.get('val_losses', [])
    val_snrs = checkpoint.get('val_snrs', [])
    val_psnrs = checkpoint.get('val_psnrs', [])
    config = checkpoint.get('config', {})
    iteration = checkpoint.get('iteration', 0)

    # Validate data
    if not train_losses:
        print("⚠️ No training data found in checkpoint!")
        return

    # Calculate validation iteration indices
    val_iters = np.arange(VALIDATION_INTERVAL,
                          VALIDATION_INTERVAL * (len(val_losses) + 1),
                          VALIDATION_INTERVAL)

    # Create figure with better layout
    fig = plt.figure(figsize=(18, 10))
    gs = fig.add_gridspec(2, 3, hspace=0.3, wspace=0.3)

    # Main title
    fig.suptitle(f'Training Metrics Dashboard - Iteration {iteration:,}\n'
                 f'Model: EdgePreservingDenoiser | '
                 f'Noise: {config.get("noise_fn", "N/A")} | '
                 f'Loss: {config.get("loss_fn", "N/A")} | '
                 f'LR: {config.get("learning_rate", "N/A")}',
                 fontsize=16, fontweight='bold', y=0.98)

    # ========== Plot 1: Training Loss (Full) ==========
    ax1 = fig.add_subplot(gs[0, 0])
    ax1.plot(train_losses, linewidth=0.5, alpha=0.6, color='steelblue', label='Raw Loss')

    # Moving average for smoother visualization
    window = min(100, max(10, len(train_losses) // 20))
    if len(train_losses) > window:
        train_ma = np.convolve(train_losses, np.ones(window) / window, mode='valid')
        ax1.plot(range(window - 1, len(train_losses)), train_ma,
                 linewidth=2, color='darkblue', label=f'Moving Avg ({window})')

    ax1.set_xlabel('Iteration', fontsize=10)
    ax1.set_ylabel('Loss', fontsize=10)
    ax1.set_title('Training Loss (Full History)', fontsize=11, fontweight='bold')
    ax1.legend(loc='upper right', fontsize=9)
    ax1.grid(True, alpha=0.3)
    ax1.set_yscale('log')

    # Add min annotation
    min_loss = min(train_losses)
    ax1.axhline(y=min_loss, color='green', linestyle='--', alpha=0.3, linewidth=1)
    ax1.text(len(train_losses) * 0.98, min_loss, f'Min: {min_loss:.6f}',
             ha='right', va='bottom', fontsize=8, color='green')

    # ========== Plot 2: Training Loss (Recent) ==========
    ax2 = fig.add_subplot(gs[0, 1])
    recent_n = min(1000, len(train_losses))
    recent_losses = train_losses[-recent_n:]
    ax2.plot(range(len(train_losses) - recent_n, len(train_losses)),
             recent_losses, linewidth=1, color='steelblue')
    ax2.set_xlabel('Iteration', fontsize=10)
    ax2.set_ylabel('Loss', fontsize=10)
    ax2.set_title(f'Training Loss (Last {recent_n})', fontsize=11, fontweight='bold')
    ax2.grid(True, alpha=0.3)

    # ========== Plot 3: Validation Loss ==========
    ax3 = fig.add_subplot(gs[0, 2])
    if val_losses:
        ax3.plot(val_iters, val_losses, marker='o', linewidth=2,
                 markersize=4, color='coral', label='Val Loss')
        ax3.set_xlabel('Iteration', fontsize=10)
        ax3.set_ylabel('Loss', fontsize=10)
        ax3.set_title('Validation Loss', fontsize=11, fontweight='bold')
        ax3.grid(True, alpha=0.3)
        ax3.set_yscale('log')

        # Mark best
        min_val_loss = min(val_losses)
        min_idx = val_losses.index(min_val_loss)
        ax3.plot(val_iters[min_idx], min_val_loss, 'r*', markersize=12,
                 label=f'Best: {min_val_loss:.6f}')
        ax3.legend(fontsize=9)
    else:
        ax3.text(0.5, 0.5, 'No validation data', ha='center', va='center',
                 transform=ax3.transAxes, fontsize=12)
        ax3.set_title('Validation Loss', fontsize=11, fontweight='bold')

    # ========== Plot 4: SNR ==========
    ax4 = fig.add_subplot(gs[1, 0])
    if val_snrs:
        ax4.plot(val_iters, val_snrs, marker='s', linewidth=2,
                 markersize=4, color='green', label='Val SNR')
        ax4.set_xlabel('Iteration', fontsize=10)
        ax4.set_ylabel('SNR (dB)', fontsize=10)
        ax4.set_title('Signal-to-Noise Ratio', fontsize=11, fontweight='bold')
        ax4.grid(True, alpha=0.3)

        # Mark best
        max_snr = max(val_snrs)
        max_idx = val_snrs.index(max_snr)
        ax4.axhline(y=max_snr, color='red', linestyle='--', alpha=0.3, linewidth=1)
        ax4.plot(val_iters[max_idx], max_snr, 'r*', markersize=12,
                 label=f'Best: {max_snr:.2f} dB')
        ax4.legend(fontsize=9)
    else:
        ax4.text(0.5, 0.5, 'No SNR data', ha='center', va='center',
                 transform=ax4.transAxes, fontsize=12)
        ax4.set_title('Signal-to-Noise Ratio', fontsize=11, fontweight='bold')

    # ========== Plot 5: PSNR ==========
    ax5 = fig.add_subplot(gs[1, 1])
    if val_psnrs:
        ax5.plot(val_iters, val_psnrs, marker='d', linewidth=2,
                 markersize=4, color='purple', label='Val PSNR')
        ax5.set_xlabel('Iteration', fontsize=10)
        ax5.set_ylabel('PSNR (dB)', fontsize=10)
        ax5.set_title('Peak Signal-to-Noise Ratio', fontsize=11, fontweight='bold')
        ax5.grid(True, alpha=0.3)

        # Mark best
        max_psnr = max(val_psnrs)
        max_idx = val_psnrs.index(max_psnr)
        ax5.axhline(y=max_psnr, color='red', linestyle='--', alpha=0.3, linewidth=1)
        ax5.plot(val_iters[max_idx], max_psnr, 'r*', markersize=12,
                 label=f'Best: {max_psnr:.2f} dB')
        ax5.legend(fontsize=9)
    else:
        ax5.text(0.5, 0.5, 'No PSNR data', ha='center', va='center',
                 transform=ax5.transAxes, fontsize=12)
        ax5.set_title('Peak Signal-to-Noise Ratio', fontsize=11, fontweight='bold')

    # ========== Plot 6: Statistics Summary ==========
    ax6 = fig.add_subplot(gs[1, 2])
    ax6.axis('off')

    stats_text = "📊 TRAINING STATISTICS\n" + "=" * 35 + "\n\n"

    stats_text += f"📈 Training Loss:\n"
    stats_text += f"   Min:     {min(train_losses):.6f}\n"
    stats_text += f"   Max:     {max(train_losses):.6f}\n"
    stats_text += f"   Final:   {train_losses[-1]:.6f}\n"
    stats_text += f"   Samples: {len(train_losses):,}\n\n"

    if val_losses:
        stats_text += f"📉 Validation Loss:\n"
        stats_text += f"   Min:     {min(val_losses):.6f}\n"
        stats_text += f"   Final:   {val_losses[-1]:.6f}\n\n"

    if val_snrs:
        stats_text += f"🔊 Val SNR:\n"
        stats_text += f"   Max:     {max(val_snrs):.2f} dB\n"
        stats_text += f"   Final:   {val_snrs[-1]:.2f} dB\n\n"

    if val_psnrs:
        stats_text += f"📸 Val PSNR:\n"
        stats_text += f"   Max:     {max(val_psnrs):.2f} dB\n"
        stats_text += f"   Final:   {val_psnrs[-1]:.2f} dB\n\n"

    stats_text += f"⚙️  Configuration:\n"
    stats_text += f"   Batch Size:  {config.get('batch_size', 'N/A')}\n"
    stats_text += f"   Learn Rate:  {config.get('learning_rate', 'N/A')}\n"
    stats_text += f"   Train Size:  {config.get('train_size', 'N/A')}\n"
    stats_text += f"   Val Size:    {config.get('val_size', 'N/A')}\n"

    ax6.text(0.05, 0.95, stats_text, transform=ax6.transAxes,
             fontsize=9, verticalalignment='top', fontfamily='monospace',
             bbox=dict(boxstyle='round', facecolor='wheat', alpha=0.3))

    plt.tight_layout()

    if save_fig:
        plot_path = checkpoint_dir / f'training_metrics_iter{iteration}.png'
        plt.savefig(plot_path, dpi=150, bbox_inches='tight')
        print(f"📊 Plot saved to {plot_path}")

    plt.show()

    # Print detailed summary to console
    print("\n" + "=" * 80)
    print("📈 DETAILED TRAINING SUMMARY")
    print("=" * 80)

    print(f"Training Loss:")
    print(f"  Min:         {min(train_losses):.6f}")
    print(f"  Max:         {max(train_losses):.6f}")
    print(f"  Final:       {train_losses[-1]:.6f}")
    print(f"  Mean:        {np.mean(train_losses):.6f}")
    print(f"  Std:         {np.std(train_losses):.6f}")
    print(f"  Samples:     {len(train_losses):,}")
    print()

    if val_losses:
        print(f"Validation Loss:")
        print(f"  Min:         {min(val_losses):.6f}")
        print(f"  Final:       {val_losses[-1]:.6f}")
        improvement = ((val_losses[0] - val_losses[-1]) / val_losses[0] * 100) if val_losses[0] != 0 else 0
        print(f"  Improvement: {improvement:.2f}%")
        print()

    if val_snrs:
        print(f"Validation SNR:")
        print(f"  Max:         {max(val_snrs):.2f} dB")
        print(f"  Final:       {val_snrs[-1]:.2f} dB")
        print(f"  Mean:        {np.mean(val_snrs):.2f} dB")
        print()

        print(f"Validation PSNR:")
        print(f"  Max:         {max(val_psnrs):.2f} dB")
        print(f"  Final:       {val_psnrs[-1]:.2f} dB")
        print(f"  Mean:        {np.mean(val_psnrs):.2f} dB")
        print()

    print("=" * 80)

def nn_test_single(model, img_np, noise_fn, noise_level, device):
    model.eval()
    noisy_np = noise_fn(img_np, noise_level)

    clean_t = torch.tensor(img_np.transpose(2, 0, 1), dtype=torch.float32).unsqueeze(0).to(device) / 255.0
    noisy_t = torch.tensor(noisy_np.transpose(2, 0, 1), dtype=torch.float32).unsqueeze(0).to(device) / 255.0

    with torch.no_grad():
        out_t = model(noisy_t).clamp(0.0, 1.0)

    clean_b = clean_t.squeeze(0).cpu()
    noisy_b = noisy_t.squeeze(0).cpu()
    out_b = out_t.squeeze(0).cpu()

    psnr_noisy = calculate_psnr(clean_b, noisy_b, max_value=1.0).item()
    snr_noisy = calculate_snr(clean_b, noisy_b).item()
    psnr_denoised = calculate_psnr(clean_b, out_b, max_value=1.0).item()
    snr_denoised = calculate_snr(clean_b, out_b).item()

    metrics = {
        "psnr_noisy": psnr_noisy,
        "snr_noisy": snr_noisy,
        "psnr_denoised": psnr_denoised,
        "snr_denoised": snr_denoised
    }

    out_np = (out_b.numpy().transpose(1, 2, 0) * 255.0).round().astype(np.uint8)

    return img_np, noisy_np, out_np, metrics


def evaluate_on_generated_images(model, num_images=1000, noise_fn=noise_mul,
                                 img_size=256, device='cuda'):
    """ 
    Evaluate model on generated synthetic images. 
 
    Args: 
        model: Trained denoising model 
        num_images: Number of images to test on 
        noise_fn: Noise function to apply 
        img_size: Size of generated images 
        device: Device to run on 
        show_plots: Whether to show distribution plots 
 
    Returns: 
        dict: Dictionary with average metrics 
    """
    model.eval()

    psnr_noisy_list = []
    psnr_denoised_list = []
    snr_noisy_list = []
    snr_denoised_list = []

    print(f"Testing on {num_images} generated images...")

    with torch.no_grad():
        for i in range(num_images):
            if (i + 1) % 100 == 0:
                print(f"  Progress: {i + 1}/{num_images}")

            # Generate image
            img = generate_image(img_size, img_size, 5, 5, 5, 5)
            noise_level = np.random.uniform(0.1, 0.3)

            # Test
            clean, noisy, denoised, metrics = nn_test_single(
                model, img, noise_fn, noise_level, device
            )

            psnr_noisy_list.append(metrics["psnr_noisy"])
            psnr_denoised_list.append(metrics["psnr_denoised"])
            snr_noisy_list.append(metrics["snr_noisy"])
            snr_denoised_list.append(metrics["snr_denoised"])

    # Calculate statistics
    results = {
        'psnr_noisy_mean': np.mean(psnr_noisy_list),
        'psnr_noisy_std': np.std(psnr_noisy_list),
        'psnr_denoised_mean': np.mean(psnr_denoised_list),
        'psnr_denoised_std': np.std(psnr_denoised_list),
        'snr_noisy_mean': np.mean(snr_noisy_list),
        'snr_noisy_std': np.std(snr_noisy_list),
        'snr_denoised_mean': np.mean(snr_denoised_list),
        'snr_denoised_std': np.std(snr_denoised_list),
        'psnr_improvement': np.mean(psnr_denoised_list) - np.mean(psnr_noisy_list),
        'snr_improvement': np.mean(snr_denoised_list) - np.mean(snr_noisy_list),
        'psnr_noisy_list': psnr_noisy_list,
        'psnr_denoised_list': psnr_denoised_list,
        'snr_noisy_list': snr_noisy_list,
        'snr_denoised_list': snr_denoised_list
    }

    # Print results
    print(f"\n{'=' * 70}")
    print(f"EVALUATION RESULTS - GENERATED IMAGES ({num_images} samples)")
    print(f"{'=' * 70}")
    print(f"Average PSNR (noisy):     {results['psnr_noisy_mean']:.2f} ± {results['psnr_noisy_std']:.2f} dB")
    print(f"Average PSNR (denoised):  {results['psnr_denoised_mean']:.2f} ± {results['psnr_denoised_std']:.2f} dB")
    print(f"Average SNR  (noisy):     {results['snr_noisy_mean']:.2f} ± {results['snr_noisy_std']:.2f} dB")
    print(f"Average SNR  (denoised):  {results['snr_denoised_mean']:.2f} ± {results['snr_denoised_std']:.2f} dB")
    print(f"{'=' * 70}")
    print(f"PSNR improvement:         {results['psnr_improvement']:.2f} dB")
    print(f"SNR improvement:          {results['snr_improvement']:.2f} dB")
    print(f"{'=' * 70}\n")

    return results

def visualize_denoising_samples(model, num_samples=5, noise_fn=noise_mul,
                                img_size=256, device='cuda'):
    """ 
    Visualize denoising results on sample images. 
 
    Args: 
        model: Trained denoising model 
        num_samples: Number of test samples to display 
        noise_fn: Noise function to apply 
        img_size: Size of generated images 
        device: Device to run on 
 
    Returns: 
        dict: Dictionary with metrics lists 
    """
    model.eval()

    # Handle plot layout
    if num_samples == 1:
        fig, axes = plt.subplots(1, 3, figsize=(12, 4))
        axes = [axes]
    else:
        fig, axes = plt.subplots(num_samples, 3, figsize=(12, 4 * num_samples))

    psnr_noisy_list = []
    psnr_denoised_list = []
    snr_noisy_list = []
    snr_denoised_list = []

    with torch.no_grad():
        for i in range(num_samples):
            # Generate image and add noise
            img = generate_image(img_size, img_size, 5, 5, 5, 5)
            noise_level = np.random.uniform(0.1, 0.3)

            # Test denoising
            clean, noisy, denoised, metrics = nn_test_single(
                model, img, noise_fn, noise_level, device
            )

            # Collect metrics
            psnr_noisy_list.append(metrics["psnr_noisy"])
            psnr_denoised_list.append(metrics["psnr_denoised"])
            snr_noisy_list.append(metrics["snr_noisy"])
            snr_denoised_list.append(metrics["snr_denoised"])

            # Plot images
            ax0, ax1, ax2 = axes[i] if num_samples > 1 else axes[0]

            ax0.imshow(clean)
            ax0.set_title(f"Clean\nNoise Level: {noise_level:.3f}")
            ax0.axis("off")

            ax1.imshow(noisy)
            ax1.set_title(f"Noisy\nPSNR: {metrics['psnr_noisy']:.2f} dB\nSNR: {metrics['snr_noisy']:.2f} dB")
            ax1.axis("off")

            ax2.imshow(denoised)
            ax2.set_title(f"Denoised\nPSNR: {metrics['psnr_denoised']:.2f} dB\nSNR: {metrics['snr_denoised']:.2f} dB")
            ax2.axis("off")

    plt.tight_layout()
    plt.show()

    # Return metrics for further analysis if needed
    return {
        'psnr_noisy_list': psnr_noisy_list,
        'psnr_denoised_list': psnr_denoised_list,
        'snr_noisy_list': snr_noisy_list,
        'snr_denoised_list': snr_denoised_list
    }

def visualize_filter_weights(model, input_image, device):
    model.eval()
    with torch.no_grad():
        x = torch.tensor(input_image.transpose(2, 0, 1), dtype=torch.float32).unsqueeze(0).to(device) / 255.0
        filtered = model.filter_branch(x)  # (1, 3, 8, H, W)
        weights = model.weight_branch(x)[0]  # (8, H, W)

    R_filtered = filtered[0, 0]

    fig, axes = plt.subplots(3, 3, figsize=(10, 10))
    axes = axes.ravel()
    for i in range(8):
        weighted_response = R_filtered[i].cpu().numpy() * weights[i].cpu().numpy()
        axes[i].imshow(weighted_response, cmap='gray')
        axes[i].set_title(f'filter {i} × weight {i}')
        axes[i].axis('off')
    axes[8].axis('off')
    plt.tight_layout()

def plot_all_channel_filters(model):
    """ 
    Display all learned filters from the FilterBranch for R, G, B channels. 
 
    Args: 
        model: Model containing filter_branch with R_conv, G_conv, B_conv 
    """
    fb = model.filter_branch

    # Extract weights for all channels
    channels = {
        'R': fb.R_conv.weight.data.cpu().squeeze(1),  # (n_filters, 11, 11)
        'G': fb.G_conv.weight.data.cpu().squeeze(1),
        'B': fb.B_conv.weight.data.cpu().squeeze(1)
    }

    # Get number of filters (assuming same for all channels)
    n_filters = channels['R'].shape[0]

    # Create figure with 3 rows (one per channel) - add extra width for labels
    fig, axes = plt.subplots(3, n_filters, figsize=(n_filters * 1.5 + 2, 6))

    # Adjust layout to make room for labels on the left
    plt.subplots_adjust(left=0.08, right=0.98, top=0.93, bottom=0.05)

    # Channel colors for labels
    channel_colors = {'R': 'red', 'G': 'green', 'B': 'blue'}
    channel_names = {'R': 'Red Channel', 'G': 'Green Channel', 'B': 'Blue Channel'}

    # Plot filters for each channel
    for row_idx, (channel_name, weights) in enumerate(channels.items()):
        for col_idx in range(n_filters):
            ax = axes[row_idx, col_idx]
            ax.imshow(weights[col_idx], cmap='gray', interpolation='nearest')
            ax.axis('off')

            # Add filter number only to top row
            if row_idx == 0:
                ax.set_title(f'F{col_idx}', fontsize=9, pad=5)

        # Add color-coded text label on the left side using fig.text
        fig.text(0.02, 0.83 - (row_idx * 0.315), channel_names[channel_name],
                 fontsize=13,
                 color=channel_colors[channel_name],
                 fontweight='bold',
                 va='center',
                 rotation=90)

    plt.suptitle('Learned Filters per Channel', fontsize=14, fontweight='bold')
    plt.show()

    # Return all filters as dictionary
    return {k: v.numpy() for k, v in channels.items()}

import os
from PIL import Image
import torch
import torchvision.transforms as transforms

def load_and_prepare_image(image_path, target_size=256):
    """ 
    Load image and prepare it for the model. 
 
    Args: 
        image_path: Path to the image file 
        target_size: Target size for center crop (default 256) 
 
    Returns: 
        tuple: (original_pil_image, model_ready_tensor) 
    """
    # Load image
    img = Image.open(image_path).convert('RGB')

    # Center crop to target size
    transform = transforms.Compose([
        transforms.CenterCrop(target_size),
        transforms.ToTensor()  # Converts to [0, 1] range
    ])

    img_tensor = transform(img)

    # Get cropped PIL image for visualization
    img_cropped = transforms.CenterCrop(target_size)(img)

    return img_cropped, img_tensor

def process_real_image(model, image_path, device='cuda'):
    """ 
    Process a real image through the denoising model. 
 
    Args: 
        model: Trained denoising model 
        image_path: Path to image 
        device: Device to run on 
 
    Returns: 
        tuple: (original_np, processed_np) 
    """
    model.eval()

    # Load and prepare image
    img_pil, img_tensor = load_and_prepare_image(image_path)

    # Convert to numpy for visualization (H, W, C)
    original_np = np.array(img_pil) / 255.0

    # Add batch dimension and move to device
    img_batch = img_tensor.unsqueeze(0).to(device)

    with torch.no_grad():
        # Process through model
        processed = model(img_batch)

        # Convert back to numpy (H, W, C)
        processed_np = processed.squeeze(0).cpu().numpy().transpose(1, 2, 0)
        processed_np = np.clip(processed_np, 0, 1)

    return original_np, processed_np

def visualize_real_image_results(model, image_paths, device='cuda'):
    """ 
    Visualize original vs processed images for multiple real images. 
 
    Args: 
        model: Trained denoising model 
        image_paths: List of image paths to process 
        device: Device to run on 
    """
    n_images = len(image_paths)
    fig, axes = plt.subplots(n_images, 2, figsize=(10, 5 * n_images))

    if n_images == 1:
        axes = [axes]

    for i, img_path in enumerate(image_paths):
        original, processed = process_real_image(model, img_path, device)

        # Get filename for title
        filename = os.path.basename(img_path)

        # Plot original
        axes[i][0].imshow(original)
        axes[i][0].set_title(f'Original\n{filename}', fontsize=10)
        axes[i][0].axis('off')

        # Plot processed
        axes[i][1].imshow(processed)
        axes[i][1].set_title(f'Model Output\n{filename}', fontsize=10)
        axes[i][1].axis('off')

    plt.suptitle('Real Image Processing Results', fontsize=14, fontweight='bold')
    plt.tight_layout()
    plt.show()

def visualize_detail_comparison(model, image_path, crop_regions, device='cuda'):
    """ 
    Compare details in specific regions (original vs processed). 
 
    Args: 
        model: Trained denoising model 
        image_path: Path to image 
        crop_regions: List of (x, y, width, height) tuples for regions to zoom into 
        device: Device to run on 
    """
    original, processed = process_real_image(model, image_path, device)

    n_regions = len(crop_regions)
    fig, axes = plt.subplots(n_regions, 3, figsize=(15, 5 * n_regions))

    if n_regions == 1:
        axes = [axes]

    for i, (x, y, w, h) in enumerate(crop_regions):
        # Full image with rectangle
        axes[i][0].imshow(original)
        axes[i][0].add_patch(plt.Rectangle((x, y), w, h,
                                           fill=False, edgecolor='red', linewidth=2))
        axes[i][0].set_title('Full Image (Region Marked)', fontsize=10)
        axes[i][0].axis('off')

        # Cropped original
        original_crop = original[y:y+h, x:x+w]
        axes[i][1].imshow(original_crop)
        axes[i][1].set_title(f'Original Detail\nRegion {i+1}', fontsize=10)
        axes[i][1].axis('off')

        # Cropped processed
        processed_crop = processed[y:y+h, x:x+w]
        axes[i][2].imshow(processed_crop)
        axes[i][2].set_title(f'Model Output Detail\nRegion {i+1}', fontsize=10)
        axes[i][2].axis('off')

    filename = os.path.basename(image_path)
    plt.suptitle(f'Detail Comparison: {filename}', fontsize=14, fontweight='bold')
    plt.tight_layout()
    plt.show()
